<script>

const canvas = document.getElementById('mainCanvas');

const ctx = canvas.getContext('2d');

const video = document.getElementById('video');

const statusEl = document.getElementById('status');

const smileValEl = document.getElementById('smileVal');

const particleCountEl = document.getElementById('particleCount');

const startBtn = document.getElementById('startBtn');

const stopBtn = document.getElementById('stopBtn');

const useFineCollisionCb = document.getElementById('useFineCollision');

const drawContourCb = document.getElementById('drawContour');


let W,H;

let camera = null;

let running = false;

let landmarker = null;


const MAX_PARTICLES = 180;

const SMILE_THRESHOLD = 0.42;

const LAUGH_THRESHOLD = 0.72;

const SMOOTH_FACTOR = 0.84;

const TYPE_RAIN = 0;

const TYPE_FIREWORK = 1;


let particles = [];

let faceBox = null;

let faceContour = [];

let faceRadius = 0;

let rawSmile = 0;

let smoothSmile = 0;


const faceContourIndices = [

  10, 338, 297, 332, 284, 251, 389, 356, 454, 323, 361, 288,

  397, 365, 379, 378, 400, 377, 152, 148, 176, 149, 150, 136,

  172, 58, 132, 93, 234, 127, 162, 21, 54, 103, 67, 109

];


function resize(){

  W = window.innerWidth;

  H = window.innerHeight;

  canvas.width = W;

  canvas.height = H;

}

resize();

window.addEventListener('resize',resize);


class Particle{

  constructor(type,x,y){

    this.type = type;

    this.x = x;

    this.y = y;

    this.r = Math.random() * 2 + 1;

    this.vx = (Math.random()-0.5)*2;

    this.vy = type === TYPE_RAIN ? Math.random()*3+2 : (Math.random()-0.5)*6;

    this.alpha = 1;

    this.decay = type === TYPE_RAIN ? 0.002 : 0.018;

  }

  update(){

    this.x += this.vx;

    this.y += this.vy;

    this.alpha -= this.decay;

    if(faceBox){

      const hx = (faceBox.x + faceBox.w/2)*W;

      const hy = (faceBox.y + faceBox.h/2)*H;

      const dx = this.x - hx;

      const dy = this.y - hy;

      const dist2 = dx*dx + dy*dy;

      const hitRange = (faceRadius * Math.min(W,H) + 80);

      if(dist2 < hitRange * hitRange){

        checkHeadCollision(this);

      }

    }

    return this.alpha > 0;

  }

  draw(){

    ctx.save();

    ctx.globalAlpha = this.alpha;

    if(this.type === TYPE_RAIN){

      ctx.strokeStyle = `rgba(150,200,255,${this.alpha})`;

      ctx.lineWidth = 1;

      ctx.beginPath();

      ctx.moveTo(this.x,this.y);

      ctx.lineTo(this.x,this.y+6);

      ctx.stroke();

    }else{

      ctx.fillStyle = `hsl(${Math.random()*360},85%,65%)`;

      ctx.beginPath();

      ctx.arc(this.x,this.y,this.r,0,Math.PI*2);

      ctx.fill();

    }

    ctx.restore();

  }

}


function checkHeadCollision(p){

  if(!faceBox || !faceRadius) return;

  const hx = (faceBox.x + faceBox.w/2)*W;

  const hy = (faceBox.y + faceBox.h/2)*H;

  const screenR = faceRadius * Math.min(W,H);

  const dx = p.x - hx;

  const dy = p.y - hy;

  const dist = Math.hypot(dx,dy);

  if(dist < screenR + p.r){

    const overlap = screenR + p.r - dist;

    const nx = dist>0 ? dx/dist : 0;

    const ny = dist>0 ? dy/dist : 0;

    p.x += nx * overlap;

    p.y += ny * overlap;

    const dot = p.vx * nx + p.vy * ny;

    p.vx = (p.vx - 2 * dot * nx) * 0.75;

    p.vy = (p.vy - 2 * dot * ny) * 0.75;

    p.vx *= 0.9;

    p.vy *= 0.9;

  }

  if(!useFineCollisionCb.checked || !faceContour.length) return;

  for(const pt of faceContour){

    const sx = pt.x * W;

    const sy = pt.y * H;

    const cdx = p.x - sx;

    const cdy = p.y - sy;

    const cdist = Math.hypot(cdx,cdy);

    if(cdist < p.r + 3){

      const overlap = p.r + 3 - cdist;

      const ncx = cdist>0 ? cdx/cdist : 0;

      const ncy = cdist>0 ? cdy/cdist : 0;

      p.x += ncx * overlap;

      p.y += ncy * overlap;

      const cdot = p.vx * ncx + p.vy * ncy;

      p.vx = (p.vx - 2 * cdot * ncx) * 0.7;

      p.vy = (p.vy - 2 * cdot * ncy) * 0.7;

    }

  }

}


function spawnParticles(){

  if(particles.length >= MAX_PARTICLES) return;

  const count = smoothSmile > LAUGH_THRESHOLD ? 6 : 2;

  const t = smoothSmile > LAUGH_THRESHOLD ? TYPE_FIREWORK : TYPE_RAIN;

  for(let i=0;i<count;i++){

    const sx = Math.random()*W;

    const sy = t===TYPE_FIREWORK ? H*0.4 + Math.random()*H*0.3 : -10;

    particles.push(new Particle(t,sx,sy));

  }

}


function drawLoop(){

  if(!running) return;

  requestAnimationFrame(drawLoop);

  ctx.clearRect(0,0,W,H);

  particles = particles.filter(p=>{

    const alive = p.update();

    if(alive) p.draw();

    return alive;

  });

  particleCountEl.textContent = particles.length;

  if(drawContourCb.checked && faceContour.length){

    ctx.save();

    ctx.strokeStyle = "rgba(126, 232, 167, 0.5)";

    ctx.lineWidth = 1.2;

    ctx.beginPath();

    faceContour.forEach((p,idx)=>{

      const sx = p.x * W;

      const sy = p.y * H;

      idx===0 ? ctx.moveTo(sx,sy) : ctx.lineTo(sx,sy);

    })

    ctx.closePath();

    ctx.stroke();

    ctx.restore();

  }

  if(smoothSmile>SMILE_THRESHOLD){

    spawnParticles();

  }

}


function onResults(res){

  if(!res.faceLandmarks || res.faceLandmarks.length===0){

    faceBox = null;

    faceContour = [];

    rawSmile = 0;

    statusEl.textContent = "未检测到人脸";

    return;

  }

  statusEl.textContent = "检测到人脸";

  const lm = res.faceLandmarks[0];

  const leftMouth = lm[61];

  const rightMouth = lm[291];

  const lipTop = lm[13];

  const lipBot = lm[14];

  const mouthWidth = Math.hypot(rightMouth.x-leftMouth.x, rightMouth.y-leftMouth.y);

  const mouthHeight = Math.hypot(lipBot.x-lipTop.x, lipBot.y-lipTop.y);

  rawSmile = mouthWidth / (mouthHeight + 0.0001);

  rawSmile = Math.min(1, Math.max(0, (rawSmile - 2.2) / 3.5));

  smoothSmile = smoothSmile * SMOOTH_FACTOR + rawSmile * (1 - SMOOTH_FACTOR);

  smileValEl.textContent = smoothSmile.toFixed(2);

  let minX=1,minY=1,maxX=0,maxY=0;

  lm.forEach(p=>{

    minX=Math.min(minX,p.x);

    minY=Math.min(minY,p.y);

    maxX=Math.max(maxX,p.x);

    maxY=Math.max(maxY,p.y);

  });

  faceBox = {x:minX,y:minY,w:maxX-minX,h:maxY-minY};

  faceContour = faceContourIndices.map(i=>lm[i]);

  const lf = lm[234];

  const rf = lm[454];

  faceRadius = Math.hypot(rf.x-lf.x, rf.y-lf.y) * 0.5;

}


async function initLandmarker(){

  const landmarker = new FaceLandmarker.FaceLandmarker({

    locateFile:f=>`https://cdn.jsdelivr.net/npm/@mediapipe/face_landmarker/${f}`

  });

  await landmarker.setOptions({

    baseOptions:{modelAssetPath:"https://cdn.jsdelivr.net/npm/@mediapipe/face_landmarker/face_landmarker.task"},

    numFaces:1,

    minFaceDetectionConfidence:0.6,

    minFacePresenceConfidence:0.6,

    minTrackingConfidence:0.6

  });

  landmarker.onResults(onResults);

  return landmarker;

}


document.addEventListener('visibilitychange',()=>{

  if(document.hidden && running){

    if(camera) camera.stop();

  }else if(!document.hidden && startBtn.disabled){

    camera.start();

  }

})


landmarker = null;

startBtn.onclick = async ()=>{

  if(running) return;

  startBtn.disabled = true;

  try{

    if(!landmarker) landmarker = await initLandmarker();

    camera = new Camera(video,{

      onFrame:async ()=>{await landmarker.send({image:video})},

      width:640,height:480

    });

    await camera.start();

    running = true;

    drawLoop();

    stopBtn.disabled = false;

    statusEl.textContent = "运行中";

  }catch(e){

    console.error(e);

    statusEl.textContent = "摄像头失败："+e.message;

    startBtn.disabled = false;

  }

};


stopBtn.onclick = ()=>{

  if(camera) camera.stop();

  running = false;

  particles = [];

  faceBox = null;

  faceContour = [];

  smoothSmile = 0;

  startBtn.disabled = false;

  stopBtn.disabled = true;

  statusEl.textContent = "已停止";

  ctx.clearRect(0,0,W,H);

};

</script>
